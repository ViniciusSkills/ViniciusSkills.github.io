            } else if (settings.activeColorTarget === "endColor") {
                settings.endColor = colorName;
            } else if (settings.activeColorTarget === "h1") {
                settings.h1Color = colorName;
            } else if (settings.activeColorTarget === "h2") {
                settings.h2Color = colorName;
                settings.savedH2Color = colorName;
            }
            updateHero();
        }


        // Map Tailwind color name to CSS value
        function getColorValue(colorName) {
            if (colorName === "black") return "#000000";
            if (colorName === "white") return "#ffffff";
            const parts = colorName.split("-");
            if (parts.length === 2) {
                const family = colorFamilies.find(f => f.name === parts[0]);
                if (family && family.shades[parts[1]]) {
                    return family.shades[parts[1]];
                }
            }
            return colorName;
        }


        // 4. Update hero preview

        function updateHero() {
            readSettings();

            hero.classList.remove("hero-light", "hero-dark");
            hero.classList.add(
                settings.darkAppearance ? "hero-dark" : "hero-light"
            );

            hero.classList.toggle(
                "hero-centre",
                settings.centreText
            );

            hero.classList.toggle(
                "hero-min-height",
                settings.minimumHeight
            );

            // H1 & H2 Serif classes
            heroHeading.classList.toggle("hero-h1-serif", settings.h1Serif);
            heroSubheading.classList.toggle("hero-h2-serif", settings.h2Serif);

            // Subheading visibility & control states
            heroSubheading.hidden = !settings.showSubheading;
            
            const h2Inputs = h2ControlsGroup.querySelectorAll("input");
            h2Inputs.forEach(input => {
                input.disabled = !settings.showSubheading;
            });
            h2ControlsGroup.style.opacity = settings.showSubheading ? "1" : "0.5";
            h2TargetBtn.disabled = !settings.showSubheading;
            h2TargetBtn.style.opacity = settings.showSubheading ? "1" : "0.5";

            if (!settings.showSubheading && settings.activeColorTarget === "h2") {
                settings.activeColorTarget = settings.gradientBackground ? "startColor" : "background";
                updateTargetButtonsUI();
            }

            // Toggle gradient UI visibility
            if (settings.gradientBackground) {
                bgTargetBtn.hidden = true;
                startColorTargetBtn.hidden = false;
                endColorTargetBtn.hidden = false;
                gradientDirectionGroup.classList.remove("hidden");
                gradientDirectionGroup.classList.add("flex");

                if (settings.activeColorTarget === "background") {
                    settings.activeColorTarget = "startColor";
                    updateTargetButtonsUI();
                }
            } else {
                bgTargetBtn.hidden = false;
                startColorTargetBtn.hidden = true;
                endColorTargetBtn.hidden = true;
                gradientDirectionGroup.classList.remove("flex");
                gradientDirectionGroup.classList.add("hidden");

                if (settings.activeColorTarget === "startColor" || settings.activeColorTarget === "endColor") {
                    settings.activeColorTarget = "background";
                    updateTargetButtonsUI();
                }
            }

            // Apply interactive letter spacings live
            heroHeading.style.letterSpacing = settings.h1Spacing;
            heroSubheading.style.letterSpacing = settings.h2Spacing;

            // Apply colors & background image
            hero.style.setProperty("--hero-bg", getColorValue(settings.bgColor));
            if (settings.gradientBackground) {
                const startHex = getColorValue(settings.startColor);
                const endHex = getColorValue(settings.endColor);
                hero.style.backgroundImage = `linear-gradient(${settings.gradientDirection}, ${startHex}, ${endHex})`;
            } else {
                hero.style.backgroundImage = "none";
            }

            heroHeading.style.color = getColorValue(settings.h1Color);
            heroSubheading.style.color = getColorValue(settings.h2Color);

            // Update UI indicator text for targets
            bgCurrentColorName.textContent = settings.bgColor;
            startColorCurrentName.textContent = settings.startColor;
            endColorCurrentName.textContent = settings.endColor;
            h1CurrentColorName.textContent = settings.h1Color;
            h2CurrentColorName.textContent = settings.h2Color;

            updateSwatchHighlights();
            updateDynamicStyles();
            updateGeneratedHTML();
        }


        function updateTargetButtonsUI() {
            targetButtons.forEach(btn => {
                const target = btn.getAttribute("data-target");
                if (target === settings.activeColorTarget) {
                    btn.classList.add("active");
                } else {
                    btn.classList.remove("active");
                }
            });
        }


        function updateSwatchHighlights() {
            let activeColor = settings.bgColor;
            if (settings.activeColorTarget === "startColor") activeColor = settings.startColor;
            if (settings.activeColorTarget === "endColor") activeColor = settings.endColor;
            if (settings.activeColorTarget === "h1") activeColor = settings.h1Color;
            if (settings.activeColorTarget === "h2") activeColor = settings.h2Color;

            paletteGrid.querySelectorAll(".swatch").forEach(swatch => {
                if (swatch.getAttribute("data-color-name") === activeColor) {
                    swatch.classList.add("selected");
                } else {
                    swatch.classList.remove("selected");
                }
            });
        }

            const styleElem = document.getElementById("dynamicHeroStyles");
            let bgStyle = `background-color: ${getColorValue(settings.bgColor)} !important;`;
            if (settings.gradientBackground) {
                bgStyle = `background-image: linear-gradient(${settings.gradientDirection}, ${getColorValue(settings.startColor)}, ${getColorValue(settings.endColor)}) !important;`;
            }

            styleElem.textContent = `
        #heroHeading {
            letter-spacing: ${settings.h1Spacing};
        }
        #heroSubheading {
            letter-spacing: ${settings.h2Spacing};
        }
        #hero {
            ${bgStyle}
        }
        #heroHeading {
            color: ${getColorValue(settings.h1Color)} !important;
        }
        #heroSubheading {
            color: ${getColorValue(settings.h2Color)} !important;
        }`;
        }


        // 5. Escape special HTML characters

        function escapeHTML(text) {
            return text
                .replace(/&/g, "&amp;")
                .replace(/</g, "&lt;")
                .replace(/>/g, "&gt;");
        }

        // 7. Update manual-copy textarea

        function updateGeneratedHTML() {
            generatedHTML.value = createExportHTML();
        }


        // 8. Handle plain-text pasting

        function handlePlainTextPaste(event) {
            event.preventDefault();
            const plainText = event.clipboardData.getData("text/plain");
            const selection = window.getSelection();

            if (!selection.rangeCount) {
                return;
            }

            const range = selection.getRangeAt(0);

            if (!event.currentTarget.contains(range.commonAncestorContainer)) {
                return;
            }

            range.deleteContents();
            const textNode = document.createTextNode(plainText);
            range.insertNode(textNode);
            range.setStartAfter(textNode);
            range.collapse(true);
            selection.removeAllRanges();
            selection.addRange(range);

            updateGeneratedHTML();
        }


        // 9. Copy HTML to clipboard

        async function copyHTML() {
            const html = createExportHTML();
            generatedHTML.value = html;
            copyStatus.textContent = "";

            try {
                if (!navigator.clipboard || !navigator.clipboard.writeText) {
                    throw new Error("Clipboard API unavailable");
                }

                await navigator.clipboard.writeText(html);

                copyStatus.textContent =
                    "HTML copied—paste it into VS Code.";

            } catch (error) {
                fallbackPanel.hidden = false;
                generatedHTML.value = html;
                copyStatus.textContent =
                    "Automatic copying is unavailable. Use the labelled text area to copy your HTML manually.";
            }
        }

            readSettings();

                settings.bgColor = "slate-900";
                settings.h1Color = "slate-50";
                settings.h2Color = "slate-300";
                settings.savedH2Color = "slate-300";
            } else {
                settings.bgColor = "slate-50";
                settings.h1Color = "slate-900";
                settings.h2Color = "slate-700";
                settings.savedH2Color = "slate-700";
            }
            updateHero();
        });

        // Gradient background toggle handler
        gradientBackground.addEventListener("change", function() {
            readSettings();
            if (settings.gradientBackground) {
                settings.startColor = settings.bgColor;
                settings.endColor = "blue-500";
                settings.activeColorTarget = "startColor";
            } else {
                settings.bgColor = settings.startColor;
                settings.activeColorTarget = "background";
            }
            updateTargetButtonsUI();
            updateHero();
        });

        targetButtons.forEach(btn => {
            btn.addEventListener("click", () => {
                settings.activeColorTarget = btn.getAttribute("data-target");

                updateSwatchHighlights();
            });

        showSubheading.addEventListener("change", () => {
